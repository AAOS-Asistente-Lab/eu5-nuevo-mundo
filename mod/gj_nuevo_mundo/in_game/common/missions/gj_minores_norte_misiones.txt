gj_minores_norte_mision_ce_ciudad_de_pesas = {
	icon = generic_traditional_economy
	player_playstyle = administrative

	visible = {
		game_has_missions_enabled = yes
		gj_minores_norte_menor_ciudad_estado = yes
		NOT = { has_variable = gj_minores_norte_hecho_mision_ce }
	}
	chance = 3600
	abort = {
	}
	on_completion = {
		set_variable = gj_minores_norte_hecho_mision_ce
	}

	gj_minores_norte_ce_tarea_gremios = {
		icon = glorious_arms

		enabled = {
			has_advance = gj_minores_norte_ce_puentes_y_gremios
		}
		duration = 0
		on_completion = {
			add_prestige = prestige_weak_bonus
		}
	}

	gj_minores_norte_ce_tarea_franquicias = {
		icon = glorious_arms
		requires = { gj_minores_norte_ce_tarea_gremios }

		enabled = {
			has_advance = gj_minores_norte_ce_carta_de_franquicias
		}
		duration = 0
		on_completion = {
			add_gold = 150
			add_prestige = prestige_mild_bonus
		}
	}

	gj_minores_norte_ce_tarea_milicia = {
		icon = glorious_arms
		requires = { gj_minores_norte_ce_tarea_franquicias }
		final = yes

		enabled = {
			has_advance = gj_minores_norte_ce_milicia_de_barrio
		}
		duration = 0
		on_completion = {
			add_country_modifier = { modifier = gj_minores_norte_sello_de_la_ciudad years = 20 mode = add_and_extend }
		}
	}
}
