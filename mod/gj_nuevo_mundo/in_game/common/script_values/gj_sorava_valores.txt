# Sorava: topes de edificios (por era) y cupos de unidades.
# Cada edificio gana un nivel de tope al llegar a ciertas eras: mejoran con el tiempo, sin romperse.

gj_sorava_nivel_casa_especias = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	if = {
		limit = { current_age_or_later = { age = age_2_renaissance } }
		add = {
			desc = "age_2_renaissance"
			value = 1
		}
	}
	if = {
		limit = { current_age_or_later = { age = age_3_discovery } }
		add = {
			desc = "age_3_discovery"
			value = 1
		}
	}
	if = {
		limit = { current_age_or_later = { age = age_5_absolutism } }
		add = {
			desc = "age_5_absolutism"
			value = 1
		}
	}
	if = {
		limit = { market_access < 0.75 }
		multiply = {
			desc = "BUILDING_LEVEL_LOW_MARKET_ACCESS_PENALTY"
			value = market_access
			add = 0.25
		}
	}
	min = 1
}
