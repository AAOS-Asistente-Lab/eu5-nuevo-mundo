# Niveles maximos de los edificios laguneños: crecen con la era
gj_lag_mercado_flotante_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	if = {
		limit = { current_age_or_later = { age = age_2_renaissance } }
		add = { desc = "age_2_renaissance" value = 1 }
	}
	if = {
		limit = { current_age_or_later = { age = age_3_discovery } }
		add = { desc = "age_3_discovery" value = 1 }
	}
	if = {
		limit = { current_age_or_later = { age = age_5_absolutism } }
		add = { desc = "age_5_absolutism" value = 1 }
	}
}
