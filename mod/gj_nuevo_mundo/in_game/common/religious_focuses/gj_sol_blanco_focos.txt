# Focos religiosos del Sol Blanco (como avances, pero de religion). 8 focos; el ultimo exige tres de los anteriores.
# El avance mensual suma lo base, el poder de la corona, el gobernante y una deriva propia: estar cerca del equilibrio (yanantin entre -25 y 25) acelera todos los focos.

gj_sol_blanco_foco_calendario = {
	monthly_progress = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = 0.05
		}
		add = {
			desc = "[crown_power|e]"
			value = "estate_power(estate_type:crown_estate)"
			multiply = 0.1
		}
		if = {
			limit = { exists = ruler_or_regent }
			add = {
				desc = "ruler_or_regent_total_abilities"
				value = ruler_or_regent.total_abilities
				multiply = 0.0015
			}
		}
		add = {
			desc = "gj_sol_blanco_deriva_equilibrio"
			value = 0
			if = {
				limit = {
					yanantin > -25
					yanantin < 25
				}
				add = 0.05
			}
		}
	}

	modifier_while_progressing = {
		stability_cost_efficiency = -0.05
	}

	modifier_on_completion = {
		monthly_prestige = 0.05
	}

	ai_will_do = {
		add = 1
	}
}
