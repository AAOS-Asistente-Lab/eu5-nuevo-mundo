# Menores del sur: dos decisiones por estilo (ciudad, selva y marca).
# Todas cuestan oro o paciencia, duran unos años y no se pueden repetir mientras dura su efecto.

# ------------------------------------------------------------------ Ciudad

gj_minores_sur_abrir_la_lonja = {
	decision_category = country_specific_decisions

	potential = {
		gj_minores_sur_estilo_ciudad = yes
	}

	allow = {
		NOT = { has_country_modifier = gj_minores_sur_lonja_abierta }
		at_war = no
		gold > {
			value = monthly_income_trade_and_tax
			multiply = 3
		}
	}

	ai_will_do = {
		value = 40
	}

	option = {
		name = gj_minores_sur_abrir_la_lonja.a
		ai_chance = { base = 100 }
		effect = {
			change_gold_effect = { scale = -3 }
			add_country_modifier = {
				modifier = gj_minores_sur_lonja_abierta
				years = 8
				mode = add
			}
		}
	}
}

gj_minores_sur_alzar_la_atalaya = {
	decision_category = country_specific_decisions

	potential = {
		gj_minores_sur_estilo_marca = yes
	}

	allow = {
		gold > {
			value = monthly_income_trade_and_tax
			multiply = 3
		}
		any_owned_location = {
			OR = {
				is_port = yes
				topography = hills
				topography = mountains
			}
			NOT = { has_location_modifier = gj_minores_sur_atalaya_encendida }
		}
	}

	ai_will_do = {
		value = 20
	}

	option = {
		name = gj_minores_sur_alzar_la_atalaya.a
		ai_chance = { base = 100 }
		effect = {
			change_gold_effect = { scale = -3 }
			random_owned_location = {
				limit = {
					OR = {
						is_port = yes
						topography = hills
						topography = mountains
					}
					NOT = { has_location_modifier = gj_minores_sur_atalaya_encendida }
				}
				add_location_modifier = {
					modifier = gj_minores_sur_atalaya_encendida
					years = 40
					mode = add
				}
			}
		}
	}
}
