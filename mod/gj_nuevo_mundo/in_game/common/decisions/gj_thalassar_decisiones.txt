# THALASSAR: decisiones. Alzar la Ciudad Hundida (5 fases) + 3 de apoyo.
# Cada fase pide oro, cristales de marea y una condicion. La fase N+1 solo se ve al hecha la N.
##############################################################

gj_thalassar_alzar_fase_1 = {
	decision_category = country_specific_decisions
	only_once = yes

	potential = {
		gj_es_thalassar = yes
		NOT = { gj_thalassar_fase_al_menos = { N = 1 } }
	}

	allow = {
		stability >= 0
		gold >= 150
		gj_thalassar_tiene_cristales = { N = 5 }
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = gj_thalassar_alzar_fase_1.a
		effect = {
			add_gold = -150
			gj_thalassar_gasta_cristales = { N = 5 }
			gj_thalassar_sube_fase = yes
			add_country_modifier = {
				modifier = gj_thal_fase_1_pais
				mode = add
				years = -1
			}
			capital = {
				add_location_modifier = {
					modifier = gj_thal_fase_1_barrio
					mode = add
					years = -1
				}
				change_development = development_mild_bonus
			}
			add_prestige = prestige_mild_bonus
			trigger_event_non_silently = gj_thalassar.21
		}
	}
}
