# Decisiones de la Liga de las Mil Lagunas (4)

# 1. Caminos de hielo: abrir las rutas sobre los lagos helados durante el invierno
gj_lag_abrir_caminos_hielo = {
	decision_category = country_specific_decisions

	potential = {
		gj_es_lagunas = yes
	}

	allow = {
		NOT = { has_country_modifier = gj_lag_caminos_de_hielo }
		NOT = { has_variable = gj_lag_caminos_enfriamiento }
		gold > { value = monthly_income_trade_and_tax multiply = 1 }
	}

	ai_will_do = {
		value = 50
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	option = {
		name = gj_lag_abrir_caminos_hielo.a
		ai_chance = { value = 100 }
		effect = {
			change_gold_effect = { scale = -1 }
			add_country_modifier = {
				modifier = gj_lag_caminos_de_hielo
				months = 5
				mode = add
			}
			set_variable = {
				name = gj_lag_caminos_enfriamiento
				years = 1
			}
			# Al llegar la primavera, el hielo cede: ver gj_lagunas.3
			trigger_event_silently = {
				id = gj_lagunas.3
				months = 5
			}
		}
	}

	option = {
		name = gj_lag_abrir_caminos_hielo.b
		ai_chance = { value = 0 }
		effect = {
			add_prestige = prestige_weak_bonus
		}
	}
}

# 4. La Gran Subasta del Estrecho: sacar a subasta los permisos de paso y rivalizar con Sorava
gj_lag_gran_subasta_estrecho = {
	decision_category = country_specific_decisions
	only_once = no

	potential = {
		gj_es_lagunas = yes
		current_age_or_later = { age = age_2_renaissance }
		NOT = { has_country_modifier = gj_lag_gran_subasta }
	}

	allow = {
		NOT = { has_variable = gj_lag_subasta_enfriamiento }
		has_country_modifier = gj_lag_barquero_mercader
	}

	ai_will_do = {
		value = 40
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	option = {
		name = gj_lag_gran_subasta_estrecho.a
		ai_chance = { value = 100 }
		effect = {
			add_country_modifier = {
				modifier = gj_lag_gran_subasta
				years = 5
				mode = add
			}
			set_variable = {
				name = gj_lag_subasta_enfriamiento
				years = 15
			}
			add_prestige = prestige_mild_bonus
			add_estate_satisfaction = {
				type = estate_type:burghers_estate
				value = estate_satisfaction_mild_bonus
			}
		}
	}
}
