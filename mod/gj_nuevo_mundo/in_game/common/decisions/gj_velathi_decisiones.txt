# Decisiones de los velathi.

gj_velathi_plantar_bosque_antiguo = {
	decision_category = country_specific_decisions

	potential = {
		gj_es_velathi = yes
	}

	allow = {
		NOT = { has_variable = gj_velathi_plantando }
		gold >= root.monthly_income_trade_and_tax
		any_owned_location = {
			development < 15
			NOT = { has_location_modifier = gj_velathi_bosque_antiguo }
		}
	}

	ai_will_do = {
		value = 10
	}

	option = {
		name = gj_velathi_plantar_bosque_antiguo.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = {
				value = root.monthly_income_trade_and_tax
				multiply = -1
			}
			random_owned_location = {
				limit = {
					development < 15
					NOT = { has_location_modifier = gj_velathi_bosque_antiguo }
				}
				add_location_modifier = { modifier = gj_velathi_bosque_antiguo years = -1 mode = add }
				add_location_modifier = { modifier = gj_velathi_brote_reciente years = 15 mode = add_and_extend }
			}
			gj_velathi_karma_gran_bonus_effect = yes
			set_variable = { name = gj_velathi_plantando years = 5 }
			trigger_event_non_silently = { id = gj_velathi.16 days = 3650 }
		}
	}
}
