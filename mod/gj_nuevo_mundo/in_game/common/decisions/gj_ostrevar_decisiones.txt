# Decisiones de Ostrevar.

gj_ostrevar_llamada_activar = {
	decision_category = country_specific_decisions
	potential = { gj_es_ostrevar = yes }
	allow = {
		NOT = { has_variable = gj_ostrevar_llamada }
		stability >= 0
	}
	ai_will_do = { value = 20 }
	option = {
		name = gj_ostrevar_llamada_activar.a
		ai_chance = { value = 50 }
		effect = {
			set_variable = gj_ostrevar_llamada
			add_country_modifier = { modifier = gj_ostrevar_llamada_ciudades mode = add }
			add_prestige = prestige_weak_bonus
		}
	}
}
