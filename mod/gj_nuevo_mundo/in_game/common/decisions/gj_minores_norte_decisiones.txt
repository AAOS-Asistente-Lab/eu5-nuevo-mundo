gj_minores_norte_decision_sellar_las_pesas = {
	decision_category = country_specific_decisions
	only_once = yes

	potential = {
		gj_minores_norte_menor_ciudad_estado = yes
		current_age_or_later = { age = age_2_renaissance }
	}

	allow = {
		has_advance = gj_minores_norte_ce_carta_de_franquicias
		gold >= 200
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = regular
		10 = interior
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = gj_minores_norte_decision_sellar_las_pesas.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = -200
			add_country_modifier = { modifier = gj_minores_norte_sello_de_la_ciudad years = 25 mode = add_and_extend }
			add_prestige = prestige_mild_bonus
		}
	}

	option = {
		name = gj_minores_norte_decision_sellar_las_pesas.b
		ai_chance = { value = 0 }
		effect = {
			add_stability = stability_weak_bonus
		}
	}
}
