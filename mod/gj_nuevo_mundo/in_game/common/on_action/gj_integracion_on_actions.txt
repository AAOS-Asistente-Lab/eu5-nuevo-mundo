# Cada faccion entra en su pais de partida. Las que no tienen todas sus piezas no se crean.
gj_integracion_inicio = {
	effect = {
		gj_asigna_recursos = yes
		gj_reparte_tierra = yes
		every_country = {
			limit = { gj_ostrevar_tag_inicial = yes }
			gj_fundar_faccion = { clave = ostrevar cultura = gj_ostrevari religion = gj_luz_fundadores nombre = GJ_OSTREVAR color = gj_pais_ostrevar }
		}
		every_country = {
			limit = { gj_ankaru_tag_inicial = yes }
			gj_fundar_faccion = { clave = ankaru cultura = gj_ankaruno religion = gj_sol_blanco nombre = GJ_ANKARU color = gj_pais_ankaru }
		}
		every_country = {
			limit = { gj_orunmar_tag_inicial = yes }
			gj_fundar_faccion = { clave = orunmar cultura = gj_orunmari religion = gj_dioses_hambrientos nombre = GJ_ORUNMAR color = gj_pais_orunmar }
		}
		every_country = {
			limit = { gj_durnhal_tag_inicial = yes }
			gj_fundar_faccion = { clave = durnhal cultura = gj_durnhali religion = gj_llama_profunda nombre = GJ_DURNHAL color = gj_pais_durnhal }
		}
		every_country = {
			limit = { gj_kaivan_tag_inicial = yes }
			gj_fundar_faccion = { clave = kaivan cultura = gj_kaivani religion = gj_cielo_clanes nombre = GJ_KAIVAN color = gj_pais_kaivan }
		}
		every_country = {
			limit = { gj_hrimval_tag_inicial = yes }
			gj_fundar_faccion = { clave = hrimval cultura = gj_insepulto religion = gj_escarcha_silente nombre = GJ_HRIMVAL color = gj_pais_hrimval }
		}
		every_country = {
			limit = { gj_skarren_tag_inicial = yes }
			gj_fundar_faccion = { clave = skarren cultura = gj_skarren religion = gj_espiritus_lago nombre = GJ_SKARREN color = gj_pais_skarren }
			set_religious_school = religious_school:gj_escuela_sagas
		}
		every_country = {
			limit = { gj_lagunas_tag_inicial = yes }
			gj_fundar_faccion = { clave = lagunas cultura = gj_lagunero religion = gj_espiritus_lago nombre = GJ_LAGUNAS color = gj_pais_lagunas }
			set_religious_school = religious_school:gj_escuela_barqueros
			if = {
				limit = { NOT = { government_type = government_type:republic } }
				change_government_type = government_type:republic
			}
		}
		every_country = {
			limit = { gj_sorava_tag_inicial = yes }
			gj_fundar_faccion = { clave = sorava cultura = gj_soravi religion = gj_pacto_balanza nombre = GJ_SORAVA color = gj_pais_sorava }
		}
		every_country = {
			limit = { gj_thalassar_tag_inicial = yes }
			gj_fundar_faccion = { clave = thalassar cultura = gj_talasio religion = gj_mar_dormido nombre = GJ_THALASSAR color = gj_pais_thalassar }
		}
		every_country = {
			limit = { gj_marea_negra_tag_inicial = yes }
			gj_fundar_faccion = { clave = marea_negra cultura = gj_mareano religion = gj_mar_dormido nombre = GJ_MAREA_NEGRA color = gj_pais_marea_negra }
			set_religious_school = religious_school:gj_escuela_marea_negra
		}
		every_country = {
			limit = { gj_velathi_tag_inicial = yes }
			gj_fundar_faccion = { clave = velathi cultura = gj_velathi religion = gj_senda_verde nombre = GJ_VELATHI color = gj_pais_velathi }
		}
	}
}
