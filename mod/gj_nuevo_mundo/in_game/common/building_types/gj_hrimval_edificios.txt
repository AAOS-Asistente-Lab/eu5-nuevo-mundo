# Edificios de la Corte Insepulta de Hrimval
# Cada uno mejora con las eras: el tope de niveles sube con la edad y el modificador escala con el nivel.

# El Relicario: guarda la vida del rey. Solo uno, solo en la capital. Si se pierde, llega el Deshielo.
gj_hrimval_relicario = {
	audio_tier = 4
	is_special = yes
	is_foreign = no
	pop_type = nobles
	max_levels = {
		value = 1
		if = {
			limit = { owner ?= { current_age_or_later = { age = age_3_discovery } } }
			add = 1
		}
		if = {
			limit = { owner ?= { current_age_or_later = { age = age_5_absolutism } } }
			add = 1
		}
	}

	category = government_category

	employment_size = large_noble_employment

	town = yes
	city = yes
	megalopolis = yes

	expensive = yes
	build_time = large_capital_build_time

	location_potential = {
		counts_as_capital = yes
	}

	country_potential = {
		gj_es_hrimval = yes
	}

	allow = {
		is_capital = yes
		country_has_no_other_copy_of_building = { type = gj_hrimval_relicario }
	}

	remove_if = {
		is_no_longer_capital = yes
	}

	capital_country_modifier = {
		monthly_legitimacy = 0.03
		global_manpower_modifier = 0.03
		country_cabinet_efficiency = 0.02
	}

	unique_production_methods = {
		gj_hrimval_relicario_mantenimiento = {
			icon_type = building_type
			icon = gj_hrimval_relicario
			stone = 0.10
			tools = 0.03
			category = building_maintenance
		}
	}

	construction_demand = early_capital_building_construction
}
