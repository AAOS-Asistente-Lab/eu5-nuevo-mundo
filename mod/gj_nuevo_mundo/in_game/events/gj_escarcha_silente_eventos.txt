# .1 La primera helada de la cuenta
gj_escarcha_silente.1 = {
	type = country_event
	title = gj_escarcha_silente.1.title
	desc = gj_escarcha_silente.1.desc
	outcome = neutral

	trigger = {
		religion = religion:gj_escarcha_silente
		has_doom = yes
		doom < 40
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	option = {
		name = gj_escarcha_silente.1.a
		add_doom = doom_weak_bonus
		add_legitimacy = legitimacy_weak_bonus
	}

	option = {
		name = gj_escarcha_silente.1.b
		add_doom = doom_weak_penalty
		add_country_modifier = {
			modifier = gj_escarcha_silente_hielo_en_las_venas
			years = 5
			mode = extend
		}
	}
}
