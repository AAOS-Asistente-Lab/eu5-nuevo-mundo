gj_mar_dormido.1 = {
	type = country_event
	title = gj_mar_dormido.1.title
	desc = gj_mar_dormido.1.desc
	outcome = neutral

	trigger = {
		gj_mar_es_creyente = yes
		NOT = { has_country_modifier = gj_mar_cancion_cuna }
		NOT = { has_country_modifier = gj_mar_silencio_respetuoso }
		NOT = { has_country_modifier = gj_mar_decima_al_agua }
	}

	option = {
		name = gj_mar_dormido.1.a
		change_gold_effect = { scale = -1 }
		add_rite_power = 8
		add_country_modifier = { modifier = gj_mar_cancion_cuna years = 6 mode = add_and_extend }
		ai_chance = {
			base = 2
		}
	}

	option = {
		name = gj_mar_dormido.1.b
		add_rite_power = 3
		add_country_modifier = { modifier = gj_mar_silencio_respetuoso years = 4 mode = add_and_extend }
		ai_chance = {
			base = 1
		}
	}

	option = {
		name = gj_mar_dormido.1.c
		trigger = {
			gj_mar_es_marea_negra = yes
		}
		change_gold_effect = { scale = -2 }
		add_rite_power = 10
		add_country_modifier = { modifier = gj_mar_decima_al_agua years = 6 mode = add_and_extend }
		ai_chance = {
			base = 4
		}
	}
}
