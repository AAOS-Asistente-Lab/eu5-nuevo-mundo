# .2 Los anos del mismo rey (cada 25 anos sube el cansancio del trono)
gj_velathi.2 = {
	type = country_event
	title = gj_velathi.2.title
	desc = gj_velathi.2.desc
	outcome = neutral
	illustration_tags = {
		10 = regular
		10 = interior
	}
	trigger = {
		gj_es_velathi = yes
		has_variable = gj_velathi_anyos_reinado
		OR = {
			var:gj_velathi_anyos_reinado = 25
			var:gj_velathi_anyos_reinado = 50
			var:gj_velathi_anyos_reinado = 75
			var:gj_velathi_anyos_reinado = 100
			var:gj_velathi_anyos_reinado = 125
			var:gj_velathi_anyos_reinado = 150
			var:gj_velathi_anyos_reinado = 175
			var:gj_velathi_anyos_reinado = 200
			var:gj_velathi_anyos_reinado = 225
			var:gj_velathi_anyos_reinado = 250
		}
	}
	immediate = {
		if = {
			limit = { NOT = { has_variable = gj_velathi_cansancio } }
			set_variable = { name = gj_velathi_cansancio value = 0 }
		}
		if = {
			limit = { var:gj_velathi_cansancio < 3 }
			change_variable = { name = gj_velathi_cansancio add = 1 }
		}
		if = {
			limit = { has_country_modifier = gj_velathi_cansancio_1 }
			remove_country_modifier = gj_velathi_cansancio_1
		}
		if = {
			limit = { has_country_modifier = gj_velathi_cansancio_2 }
			remove_country_modifier = gj_velathi_cansancio_2
		}
		if = {
			limit = { has_country_modifier = gj_velathi_cansancio_3 }
			remove_country_modifier = gj_velathi_cansancio_3
		}
		if = {
			limit = { var:gj_velathi_cansancio = 1 }
			add_country_modifier = { modifier = gj_velathi_cansancio_1 years = 40 mode = add_and_extend }
		}
		if = {
			limit = { var:gj_velathi_cansancio = 2 }
			add_country_modifier = { modifier = gj_velathi_cansancio_2 years = 40 mode = add_and_extend }
		}
		if = {
			limit = { var:gj_velathi_cansancio >= 3 }
			add_country_modifier = { modifier = gj_velathi_cansancio_3 years = 40 mode = add_and_extend }
		}
	}
	option = {
		name = gj_velathi.2.a
		add_prestige = prestige_mild_bonus
	}
	option = {
		name = gj_velathi.2.b
		add_prestige = prestige_mild_penalty
		gj_velathi_reinicia_cansancio_effect = yes
	}
}

# .3 El Sueno Largo
gj_velathi.3 = {
	type = country_event
	title = gj_velathi.3.title
	desc = gj_velathi.3.desc
	outcome = negative
	illustration_tags = {
		10 = regular
		10 = interior
	}
	trigger = {
		gj_es_velathi = yes
		has_variable = gj_velathi_anyos_reinado
		var:gj_velathi_anyos_reinado >= 75
		NOT = { has_country_modifier = gj_velathi_consejo_regente }
	}
	option = {
		name = gj_velathi.3.a
		add_country_modifier = { modifier = gj_velathi_consejo_regente years = 15 mode = add_and_extend }
		ruler ?= {
			add_character_modifier = { modifier = gj_velathi_rey_dormido years = 15 mode = add }
		}
		trigger_event_non_silently = { id = gj_velathi.4 days = 5475 }
	}
	option = {
		name = gj_velathi.3.b
		add_legitimacy = legitimacy_weak_penalty
		add_prestige = prestige_weak_penalty
		gj_velathi_karma_bonus_effect = yes
	}
	option = {
		name = gj_velathi.3.c
		add_country_modifier = { modifier = gj_velathi_heredera_en_el_consejo years = 10 mode = add_and_extend }
		add_stability = stability_weak_penalty
	}
}
