gj_espiritus_lago.1 = {
	type = country_event
	title = gj_espiritus_lago.1.title
	desc = gj_espiritus_lago.1.desc
	outcome = neutral

	trigger = {
		gj_espiritus_lago_es_creyente = yes
		gj_espiritus_lago_tiene_lectura_activa = no
	}

	option = {
		name = gj_espiritus_lago.1.a
		if = {
			limit = { gj_espiritus_lago_es_barquero = yes }
			add_country_modifier = { modifier = gj_espiritus_lago_lectura_mercaderes years = 8 mode = add_and_extend }
		}
		else = {
			add_country_modifier = { modifier = gj_espiritus_lago_lectura_mercaderes years = 4 mode = add_and_extend }
		}
		gj_espiritus_lago_registra_lectura = yes
		ai_chance = {
			base = 1
			modifier = {
				add = 3
				gj_espiritus_lago_es_barquero = yes
			}
		}
	}

	option = {
		name = gj_espiritus_lago.1.b
		if = {
			limit = { gj_espiritus_lago_es_saga = yes }
			add_country_modifier = { modifier = gj_espiritus_lago_lectura_incursion years = 8 mode = add_and_extend }
		}
		else = {
			add_country_modifier = { modifier = gj_espiritus_lago_lectura_incursion years = 4 mode = add_and_extend }
		}
		gj_espiritus_lago_registra_lectura = yes
		ai_chance = {
			base = 1
			modifier = {
				add = 3
				gj_espiritus_lago_es_saga = yes
			}
		}
	}

	option = {
		name = gj_espiritus_lago.1.c
		add_country_modifier = { modifier = gj_espiritus_lago_lectura_prudencia years = 5 mode = add_and_extend }
		gj_espiritus_lago_registra_lectura = yes
		ai_chance = {
			base = 1
		}
	}
}

gj_espiritus_lago.2 = {
	type = country_event
	title = gj_espiritus_lago.2.title
	desc = gj_espiritus_lago.2.desc
	outcome = negative

	trigger = {
		gj_espiritus_lago_es_creyente = yes
		NOT = { has_country_modifier = gj_espiritus_lago_aguas_sin_hielo }
	}

	option = {
		name = gj_espiritus_lago.2.a
		change_gold_effect = { scale = -2 }
		add_country_modifier = { modifier = gj_espiritus_lago_ofrenda_aceptada years = 10 mode = add_and_extend }
		gj_espiritus_lago_registra_lectura = yes
	}

	option = {
		name = gj_espiritus_lago.2.b
		add_country_modifier = { modifier = gj_espiritus_lago_aguas_sin_hielo years = 5 mode = add_and_extend }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
	}

	option = {
		name = gj_espiritus_lago.2.c
		trigger = {
			gj_espiritus_lago_es_barquero = yes
		}
		change_gold_effect = { scale = 1 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
		add_legitimacy = legitimacy_weak_penalty
	}
}
