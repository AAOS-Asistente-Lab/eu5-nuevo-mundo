namespace = gj_senda_verde

# 1. La hora de la poda: karma alto, el bosque se cierra sobre el pueblo.
gj_senda_verde.1 = {
	type = country_event
	title = gj_senda_verde.1.title
	desc = gj_senda_verde.1.desc
	outcome = neutral

	trigger = {
		religion = religion:gj_senda_verde
		has_ruler = yes
		karma > 33
		NOT = { has_variable = gj_senda_verde_cd_1 }
	}

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	immediate = {
		set_variable = { name = gj_senda_verde_cd_1 value = 1 years = 4 }
	}

	option = {
		name = gj_senda_verde.1.a
		add_karma = karma_mild_penalty
		add_country_modifier = { modifier = gj_senda_verde_poda_provechosa years = 5 mode = add_and_extend }
	}
	option = {
		name = gj_senda_verde.1.b
		add_karma = karma_weak_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
		add_stability = stability_weak_penalty
	}
	option = {
		name = gj_senda_verde.1.c
		add_karma = karma_weak_penalty
		add_legitimacy = legitimacy_weak_bonus
	}
}

# 3. Los plantadores de primavera: karma en calma.
gj_senda_verde.3 = {
	type = country_event
	title = gj_senda_verde.3.title
	desc = gj_senda_verde.3.desc
	outcome = positive

	trigger = {
		religion = religion:gj_senda_verde
		has_ruler = yes
		karma >= -33
		karma <= 33
		NOT = { has_variable = gj_senda_verde_cd_3 }
	}

	illustration_tags = {
		10 = exterior
		10 = happy
	}

	immediate = {
		set_variable = { name = gj_senda_verde_cd_3 value = 1 years = 6 }
		random_owned_location = {
			save_scope_as = target_location
		}
	}

	option = {
		name = gj_senda_verde.3.a
		add_karma = karma_weak_bonus
		scope:target_location = {
			add_location_modifier = { modifier = gj_senda_verde_brotes_de_fiesta years = 5 mode = add_and_extend }
		}
	}
	option = {
		name = gj_senda_verde.3.b
		add_country_modifier = { modifier = gj_senda_verde_cosecha_de_savia years = 5 mode = add_and_extend }
	}
}

# 5. Fuego en la linde.
gj_senda_verde.5 = {
	type = country_event
	title = gj_senda_verde.5.title
	desc = gj_senda_verde.5.desc
	outcome = negative

	trigger = {
		religion = religion:gj_senda_verde
		has_ruler = yes
		NOT = { has_variable = gj_senda_verde_cd_5 }
		any_owned_location = {
			OR = {
				vegetation = forest
				vegetation = woods
			}
		}
	}

	illustration_tags = {
		10 = exterior
		10 = angry
	}

	immediate = {
		set_variable = { name = gj_senda_verde_cd_5 value = 1 years = 6 }
		random_owned_location = {
			limit = {
				OR = {
					vegetation = forest
					vegetation = woods
				}
			}
			save_scope_as = target_location
		}
	}

	option = {
		name = gj_senda_verde.5.a

		add_prestige = prestige_weak_bonus
	}
	option = {
		name = gj_senda_verde.5.b
