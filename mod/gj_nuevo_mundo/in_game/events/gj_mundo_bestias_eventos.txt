# .1 La Durmiente se revuelve (estado 0 a 1)
gj_mundo_bestias.1 = {
	type = country_event
	title = gj_mundo_bestias.1.title
	desc = gj_mundo_bestias.1.desc
	outcome = negative

	trigger = {
		gj_mundo_bestias_dragon_dormido = yes
		gj_mundo_bestias_pais_caldera = yes
		OR = {
			gj_mundo_bestias_inquietud_caldera_alta = yes
			at_war = yes
			current_age_or_later = { age = age_4_reformation }
		}
	}

	immediate = {
		gj_mundo_bestias_dragon_pone_estado = { ESTADO = 1 }
	}

	option = {
		name = gj_mundo_bestias.1.a
		ai_chance = { base = 50 }
		add_country_modifier = { modifier = gj_mundo_bestias_dragon_alerta years = 5 mode = add_and_extend }
	}
	option = {
		name = gj_mundo_bestias.1.b
		ai_chance = { base = 30 }
		trigger = { gold >= gj_mundo_bestias_coste_aplacar }
		add_gold = {
			value = gj_mundo_bestias_coste_aplacar
			multiply = -1
		}
		add_country_modifier = { modifier = gj_mundo_bestias_dragon_aplacado years = 5 mode = add_and_extend }
		random_list = {
			40 = { gj_mundo_bestias_dragon_pone_estado = { ESTADO = 0 } }
			60 = { }
		}
	}
	option = {
		name = gj_mundo_bestias.1.c
		ai_chance = { base = 20 }
		add_gold = 60
	}
}
