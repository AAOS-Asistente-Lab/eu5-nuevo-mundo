# Velas en el horizonte: el primer contacto del pais con el Viejo Mundo.
gj_mundo_contacto.1 = {
	type = country_event
	title = gj_mundo_contacto.1.title
	desc = gj_mundo_contacto.1.desc
	outcome = neutral
	historical_info = gj_mundo_contacto.1.historical_info

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	trigger = {
		gj_contacto_posible = yes
	}

	immediate = {
		gj_contacto_marca = yes
		if = {
			limit = { exists = scope:gj_visitante }
			scope:gj_visitante = {
				trigger_event_non_silently = { id = gj_mundo_contacto.2 days = 20 }
				trigger_event_non_silently = { id = gj_mundo_contacto.19 years = 3 }
			}
		}
	}

	option = {
		name = gj_mundo_contacto.1.a
		ai_chance = { base = 50 }
		gj_contacto_elige_comerciar = yes
		gj_contacto_ensena_al_visitante = yes
		change_gold_effect = { scale = 2 }
		trigger_event_non_silently = { id = gj_mundo_contacto.3 years = 2 }
	}

	option = {
		name = gj_mundo_contacto.1.b
		ai_chance = { base = 35 }
		gj_contacto_elige_aprender = yes
		gj_contacto_ensena_al_visitante = yes
		change_gold_effect = { scale = -1 }
		trigger_event_non_silently = { id = gj_mundo_contacto.4 years = 2 }
	}

	option = {
		name = gj_mundo_contacto.1.c
		ai_chance = { base = 15 }
		gj_contacto_elige_cerrarse = yes
		trigger_event_non_silently = { id = gj_mundo_contacto.5 years = 5 }
	}
}
