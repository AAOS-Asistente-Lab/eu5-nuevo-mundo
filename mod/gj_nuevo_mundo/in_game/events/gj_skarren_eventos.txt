namespace = gj_skarren

gj_skarren.1 = {
	type = country_event
	title = gj_skarren.1.title
	desc = gj_skarren.1.desc
	outcome = neutral
	illustration_tags = {
		10 = regular
		10 = exterior
	}
	fire_only_once = yes

	trigger = {
		gj_es_skarren = yes
		NOT = { has_variable = gj_skarren_iniciado }
	}

	immediate = {
		set_variable = gj_skarren_iniciado
		set_variable = { name = gj_skarren_gloria value = 0 }
		set_variable = { name = gj_skarren_sagas value = 0 }
		set_variable = { name = gj_skarren_botin_reciente value = yes years = 6 }
	}

	option = {
		name = gj_skarren.1.a
		add_legitimacy = legitimacy_mild_bonus
			gj_skarren_suma_gloria = { N = 2 }
	}

	option = {
		name = gj_skarren.1.b
		add_prestige = prestige_mild_bonus
			add_army_tradition = army_tradition_mild_bonus
			add_stability = stability_weak_penalty
			gj_skarren_suma_gloria = { N = 3 }
	}
}
