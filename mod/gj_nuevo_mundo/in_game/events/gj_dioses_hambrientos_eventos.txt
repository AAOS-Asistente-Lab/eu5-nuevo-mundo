namespace = gj_dioses_hambrientos

# .1 Los dioses murmuran: tres anos de paz, la devocion se resiente.
gj_dioses_hambrientos.1 = {
	type = country_event
	title = gj_dioses_hambrientos.1.title
	desc = gj_dioses_hambrientos.1.desc
	outcome = neutral
	hide_portraits = yes

	trigger = {
		religion = religion:gj_dioses_hambrientos
		has_variable = gj_dioses_hambrientos_anos_de_paz
		var:gj_dioses_hambrientos_anos_de_paz >= 3
		at_war = no
		NOT = { has_variable = gj_dioses_hambrientos_murmullo_enfriamiento }
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		set_variable = { name = gj_dioses_hambrientos_murmullo_enfriamiento value = yes days = 1460 }
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
	}

	option = {
		name = gj_dioses_hambrientos.1.a
		change_gold_effect = { scale = -2 }
		add_religious_influence = religious_influence_weak_bonus
		change_variable = { name = gj_dioses_hambrientos_anos_de_paz add = -1 }
		clamp_variable = { name = gj_dioses_hambrientos_anos_de_paz min = 0 max = 100 }
	}
	option = {
		name = gj_dioses_hambrientos.1.b
		add_religious_influence = religious_influence_weak_penalty
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_penalty }
	}
	option = {
		name = gj_dioses_hambrientos.1.c
		trigger = {
			has_religious_aspect = religious_aspect:gj_dioses_hambrientos_aspecto_guardian_silos
		}
		add_legitimacy = legitimacy_weak_bonus
	}
}

	}

	}

# .5 La Cuenta se cierra: cinco anos despues del presagio.
gj_dioses_hambrientos.5 = {
	type = country_event
	title = gj_dioses_hambrientos.5.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = gj_dioses_hambrientos_guerra_en_plazo }
				desc = gj_dioses_hambrientos.5.desc.a
			}
			triggered_desc = {
				desc = gj_dioses_hambrientos.5.desc.b
			}
		}
	}
	outcome = neutral
	hide_portraits = yes

	trigger = {
		religion = religion:gj_dioses_hambrientos
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		if = {
			limit = { has_variable = gj_dioses_hambrientos_presagio_abierto }
			remove_variable = gj_dioses_hambrientos_presagio_abierto
		}
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:nobles_estate }
	}

	option = {
		name = gj_dioses_hambrientos.5.a
		trigger = {
			has_variable = gj_dioses_hambrientos_guerra_en_plazo
		}
		add_religious_influence = religious_influence_severe_bonus
		add_prestige = prestige_mild_bonus
		add_country_modifier = { modifier = gj_dioses_hambrientos_cuenta_cumplida years = 10 mode = replace }
		remove_variable = gj_dioses_hambrientos_guerra_en_plazo
	}
	option = {
		name = gj_dioses_hambrientos.5.b
		trigger = {
			NOT = { has_variable = gj_dioses_hambrientos_guerra_en_plazo }
		}
		add_religious_influence = religious_influence_severe_penalty
		add_stability = stability_weak_penalty
		add_country_modifier = { modifier = gj_dioses_hambrientos_dioses_airados years = 5 mode = replace }
	}
}

# .6 El pozo que habla: un cautivo al pozo, y el pozo responde (o calla).
gj_dioses_hambrientos.6 = {
	type = country_event
	title = gj_dioses_hambrientos.6.title
	desc = gj_dioses_hambrientos.6.desc
	outcome = neutral
	hide_portraits = yes

	trigger = {
		religion = religion:gj_dioses_hambrientos
		pop_type_population_in_country = {
			pop_type = pop_type:slaves
			value > 0.1
		}
		NOT = { has_variable = gj_dioses_hambrientos_pozo_enfriamiento }
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		set_variable = { name = gj_dioses_hambrientos_pozo_enfriamiento value = yes days = 1825 }
		event_illustration_poptype_effect = { foreground = pop_type:slaves background = pop_type:peasants }
	}

	option = {
		name = gj_dioses_hambrientos.6.a
		gj_dioses_hambrientos_consume_cautivos = { FRACCION = -0.2 }
		random_list = {
			50 = {
				add_religious_influence = religious_influence_mild_bonus
				add_prestige = prestige_weak_bonus
			}
			30 = {
				add_religious_influence = religious_influence_severe_bonus
				capital = {
					add_location_modifier = {
						modifier = gj_dioses_hambrientos_pozo_bendecido
						years = 20
						mode = add_and_extend
					}
				}
			}
			20 = {
				add_religious_influence = religious_influence_weak_penalty
				add_stability = stability_weak_penalty
			}
		}
	}
	option = {
		name = gj_dioses_hambrientos.6.b
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_penalty }
		add_legitimacy = legitimacy_weak_bonus
	}
}
