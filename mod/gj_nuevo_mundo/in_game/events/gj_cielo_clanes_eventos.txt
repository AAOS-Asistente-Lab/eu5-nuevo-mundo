# El pabellon del Cielo cae
gj_cielo_clanes.2 = {
	type = country_event
	title = gj_cielo_clanes.2.title
	desc = gj_cielo_clanes.2.desc
	outcome = negative

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	trigger = {
		religion = religion:gj_cielo_clanes
		has_variable = gj_dios_reinado
		NOT = { has_variable = gj_cd_pabellon }
	}

	immediate = {
		set_variable = {
			name = gj_cd_pabellon
			value = 1
			years = 12
		}
	}

	option = {
		name = gj_cielo_clanes.2.a
		change_gold_effect = { scale = -1 }
		add_prestige = prestige_weak_bonus
		gj_cielo_clanes_favor = { n = 1 }
		gj_cielo_clanes_recalcula_favor = yes
	}

	option = {
		name = gj_cielo_clanes.2.b
		add_country_modifier = {
			modifier = gj_pabellon_caido
			years = modifier_duration_years_normal
			mode = add_and_extend
		}
		gj_cielo_clanes_favor = { n = -1 }
		gj_cielo_clanes_recalcula_favor = yes
	}
}
