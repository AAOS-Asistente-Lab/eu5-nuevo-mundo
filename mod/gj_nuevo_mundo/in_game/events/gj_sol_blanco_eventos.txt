# El Sol demasiado alto: el yanantin se ha inclinado hacia lo solar.
gj_sol_blanco.1 = {
	type = country_event
	title = gj_sol_blanco.1.title
	desc = gj_sol_blanco.1.desc
	outcome = negative

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	trigger = {
		is_subject = no
		religion = religion:gj_sol_blanco
		yanantin < -35
	}

	option = {
		name = gj_sol_blanco.1.a
		trigger = {
			gold >= { value = monthly_income_trade_and_tax multiply = 1 }
		}
		add_gold = { value = monthly_income_trade_and_tax multiply = -1 }
		add_yanantin = 20
		add_prestige = prestige_weak_bonus
	}
	option = {
		name = gj_sol_blanco.1.b
		high_risk_option = yes
		add_legitimacy = legitimacy_weak_bonus
		add_yanantin = -5
		add_stability = stability_weak_penalty
	}
}

# El Sol se apaga: lectura de un eclipse (lo lanza gj_sol_blanco_lee_presagio).
# Sin rito, es una crisis de legitimidad; con rito, una gran bonificacion.
gj_sol_blanco.8 = {
	type = country_event
	title = gj_sol_blanco.8.title
	desc = gj_sol_blanco.8.desc
	outcome = neutral

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	trigger = {
		religion = religion:gj_sol_blanco
	}

	option = {
		name = gj_sol_blanco.8.a
		trigger = {
			gold >= { value = monthly_income_trade_and_tax multiply = 2 }
		}
		add_gold = { value = monthly_income_trade_and_tax multiply = -2 }
		add_country_modifier = {
			modifier = gj_sol_blanco_eclipse_celebrado
			years = 10
			mode = add
		}
		if = {
			limit = { yanantin > 0 }
			add_yanantin = -20
		}
		else_if = {
			limit = { yanantin < 0 }
			add_yanantin = 20
		}
	}
	option = {
		name = gj_sol_blanco.8.b
		trigger = {
			has_variable = gj_sol_blanco_vigilia_eclipse
		}
		add_country_modifier = {
			modifier = gj_sol_blanco_eclipse_celebrado
			years = 10
			mode = add
		}
	}
	option = {
		name = gj_sol_blanco.8.c
		fallback = yes
		add_country_modifier = {
			modifier = gj_sol_blanco_eclipse_ignorado
			years = 10
			mode = add
		}
		add_legitimacy = legitimacy_mild_penalty
	}
}
