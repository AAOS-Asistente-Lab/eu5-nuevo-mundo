namespace = gj_orunmar

########################################
# gj_orunmar.1  La Mesa Roja se reune (una vez, al empezar)
########################################
gj_orunmar.1 = {
	type = country_event
	title = gj_orunmar.1.title
	desc = gj_orunmar.1.desc
	outcome = neutral
	fire_only_once = yes

	trigger = {
		gj_es_orunmar = yes
	}

	option = {
		name = gj_orunmar.1.a
		random_neighbor_country = {
			limit = {
				NOT = { is_subject_of = root }
				NOT = { is_allied_with = { target = root } }
			}
			root = {
				add_casus_belli = { target = prev type = casus_belli:cb_gj_orunmar_cosecha }
			}
		}
		add_country_modifier = {
			modifier = gj_orunmar_marcha_de_cosecha
			years = 5
			mode = add_and_extend
		}
	}
	option = {
		name = gj_orunmar.1.b
		add_country_modifier = {
			modifier = gj_orunmar_niebla_bendita
			years = 8
			mode = add_and_extend
		}
	}
	option = {
		name = gj_orunmar.1.c
		add_country_modifier = {
			modifier = gj_orunmar_bonanza_de_tinte
			years = 8
			mode = add_and_extend
		}
	}
}

########################################
# gj_orunmar.11  Embajada de los hijos del Sol
########################################
gj_orunmar.11 = {
	type = country_event
	title = gj_orunmar.11.title
	desc = gj_orunmar.11.desc
	outcome = neutral

	trigger = {
		gj_es_orunmar = yes
		any_neighbor_country = {
			religion = religion:gj_sol_blanco
		}
	}

	immediate = {
		random_neighbor_country = {
			limit = {
				religion = religion:gj_sol_blanco
			}
			save_scope_as = gj_vecino_del_sol
		}
	}

	option = {
		name = gj_orunmar.11.a
		add_country_modifier = {
			modifier = gj_orunmar_paz_con_el_sol
			years = 6
			mode = add_and_extend
		}
		add_prestige = prestige_weak_bonus
	}
	option = {
		name = gj_orunmar.11.b
		add_casus_belli = { target = scope:gj_vecino_del_sol type = casus_belli:cb_gj_orunmar_cosecha }
		add_country_modifier = {
			modifier = gj_orunmar_marcha_de_cosecha
			years = 4
			mode = add_and_extend
		}
	}
}

########################################
# gj_orunmar.16  Las orugas rojas
########################################
gj_orunmar.16 = {
	type = country_event
	title = gj_orunmar.16.title
	desc = gj_orunmar.16.desc
	outcome = negative

	trigger = {
		gj_es_orunmar = yes
	}

	immediate = {
		random_owned_location = {
			limit = {
				OR = {
					topography = hills
					topography = mountains
					topography = plateau
				}
			}
			save_scope_as = gj_campo_afectado
			add_location_modifier = {
				modifier = gj_orunmar_orugas_rojas
				years = 4
				mode = add_and_extend
			}
		}
	}

	option = {
		name = gj_orunmar.16.a
		add_stability = stability_weak_penalty
	}
	option = {
		name = gj_orunmar.16.b
		add_gold = -40
	}
}
