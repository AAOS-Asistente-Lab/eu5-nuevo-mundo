# Situacion great_pestilence de mundo_contacto: «Fiebres de los barcos».
# Sustituye a la del juego base (REPLACE). Cambios:
#  - los eventos mensuales son los nuestros (gj_mundo_contacto.7 a .10), mucho mas suaves que great_pestilence.3 a .8;
#  - al acabar, el evento final es gj_mundo_contacto.11;
#  - la situacion se sigue viendo con los modificadores del juego base y con los nuestros.

REPLACE:great_pestilence = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	hint_tag = hint_great_pestilence
	can_start = {
		disease_is_active = disease:great_pestilence
	}

	outcome = {
		name = "great_pestilence_disease_exhausted"
		desc = "great_pestilence_disease_exhausted_desc"
		# Acaba cuando el brote se apaga. No hace falta que haya pasado por tres regiones concretas:
		# el mundo del mod no tiene por que repetir el reparto de la America del juego base.
		trigger = {
			NOT = {
				disease_is_active = disease:great_pestilence
			}
		}
	}

	visible = {
		OR = {
			country_has_disease = disease:great_pestilence
			has_country_modifier = no_contact_with_outsiders
			has_country_modifier = procure_remedies
			has_country_modifier = procure_remedies_cost
			has_country_modifier = segregate_the_infected
			has_country_modifier = blame_the_minorities
			has_country_modifier = "sponsor_spiritual_protection_bonus"
			has_country_modifier = sponsor_spiritual_protection_cost
			has_country_modifier = gj_ev_cuarentena_muelles
			has_country_modifier = gj_ev_medicos_del_otro_lado
			has_country_modifier = gj_ev_oracion_y_hierbas
		}
	}

	on_start = {
		set_variable = { name = var_carib_was_infected value = no }
		set_variable = { name = var_mesoamerica_was_infected value = no }
		set_variable = { name = var_andes_was_infected value = no }
	}

	on_monthly = {
		every_country = {
			limit = {
				country_has_disease = disease:great_pestilence
			}
			set_variable = { name = had_great_pestilence value = yes }
			random_list = {
				2 = { trigger_event_non_silently = gj_mundo_contacto.7 }
				2 = { trigger_event_non_silently = gj_mundo_contacto.8 }
				1 = { trigger_event_non_silently = gj_mundo_contacto.9 }
				1 = { trigger_event_non_silently = gj_mundo_contacto.10 }
				60 = { }
			}
		}
		if = {
			limit = {
				var:var_carib_was_infected = no
				region:caribbean_region = {
					any_ownable_location_in_region = {
						disease_presence = { disease = disease:great_pestilence value > 0 }
					}
				}
			}
			set_variable = { name = var_carib_was_infected value = yes }
		}
		if = {
			limit = {
				var:var_mesoamerica_was_infected = no
				region:mesoamerica_region = {
					any_ownable_location_in_region = {
						disease_presence = { disease = disease:great_pestilence value > 0 }
					}
				}
			}
			set_variable = { name = var_mesoamerica_was_infected value = yes }
		}
		if = {
			limit = {
				var:var_andes_was_infected = no
				region:andes_region = {
					any_ownable_location_in_region = {
						disease_presence = { disease = disease:great_pestilence value > 0 }
					}
				}
			}
			set_variable = { name = var_andes_was_infected value = yes }
		}
	}

	tooltip = {
		if = {
			limit = {
				disease_presence = { disease = disease:great_pestilence value > 0 }
			}
			custom_tooltip = "GREAT_PESTILENCE_PRESENT_IN_LOCATION"
		}
	}

	on_ended = {
		add_timeline_note = {
			key = timeline_note_great_pestilence_end
			show_in_pop_graph = yes
			unique = yes
		}
		every_country = {
			limit = {
				has_variable = had_great_pestilence
			}
			remove_country_modifier = no_contact_with_outsiders
			remove_country_modifier = procure_remedies
			remove_country_modifier = procure_remedies_cost
			remove_country_modifier = segregate_the_infected
			remove_country_modifier = blame_the_minorities
			remove_country_modifier = "sponsor_spiritual_protection_bonus"
			remove_country_modifier = sponsor_spiritual_protection_cost

			trigger_event_non_silently = gj_mundo_contacto.11
		}
		every_market_in_world = {
			limit = {
				location.continent = continent:america
				has_temporary_demand = demand:procure_remedies
				has_temporary_demand = demand:procure_remedies_no_liquor
			}
			remove_temporary_demand = demand:procure_remedies
			remove_temporary_demand = demand:procure_remedies_no_liquor
		}
	}

	is_data_map = yes
	map_color = {
		if = {
			limit = {
				disease_presence = { disease = disease:great_pestilence value > 0 }
			}
			lerp = {
				min_color = rgb { 248  180  194 }
				max_color = rgb { 120 20 40 }
				factor = "disease_presence(disease:great_pestilence)"
			}
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	legend_key = {
		desc = "great_pestilence"
		color = rgb { 120 20 40 }
	}
}
