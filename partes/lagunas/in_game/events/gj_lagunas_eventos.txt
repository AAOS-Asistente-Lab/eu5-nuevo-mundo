# .1 La Asamblea elige al Barquero Mayor
gj_lagunas.1 = {
	type = country_event
	title = gj_lagunas.1.title
	desc = gj_lagunas.1.desc
	outcome = neutral

	trigger = {
		gj_es_lagunas = yes
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = gj_lagunas.1.a
		ai_chance = { factor = 4 }
		add_country_modifier = { modifier = gj_lag_barquero_mercader years = 8 mode = add }
		set_variable = { name = gj_lag_mandato_barquero years = 8 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = gj_lagunas.1.b
		ai_chance = { factor = 3 }
		add_country_modifier = { modifier = gj_lag_barquero_armador years = 8 mode = add }
		set_variable = { name = gj_lag_mandato_barquero years = 8 }
	}

	option = {
		name = gj_lagunas.1.c
		ai_chance = { factor = 2 }
		add_country_modifier = { modifier = gj_lag_barquero_tejedor years = 8 mode = add }
		set_variable = { name = gj_lag_mandato_barquero years = 8 }
	}

	option = {
		name = gj_lagunas.1.d
		ai_chance = { factor = 1 }
		add_country_modifier = { modifier = gj_lag_barquero_juez years = 8 mode = add }
		set_variable = { name = gj_lag_mandato_barquero years = 8 }
		add_stability = stability_weak_bonus
	}
}
