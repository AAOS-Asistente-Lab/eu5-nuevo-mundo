# Edificios de la Liga de las Mil Lagunas (5). Todos mejoran con la era: su nivel maximo
# sube con la edad (ver script_values/gj_lagunas_valores.txt). Ninguno supera en mas de un
# nivel a su equivalente del juego base, y todos tienen un limite de lugar.

# 1. Mercado Flotante: solo junto a un lago. Mas capacidad de mercado y mas comercio por burgues.
gj_lag_mercado_flotante = {
	audio_tier = 2
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = gj_lag_mercado_flotante_max_level

	category = trade_category
	employment_size = trade_employment

	town = yes
	city = yes
	megalopolis = yes

	expensive = yes
	build_time = merchant_build_time

	country_potential = {
		gj_es_lagunas = yes
	}
	location_potential = {
		is_adjacent_to_lake = yes
	}

	modifier = {
		local_merchant_capacity = 1.5
		local_merchant_power = 0.5
		local_trades_per_burgher = 0.10
	}

	unique_production_methods = {
		gj_lag_mercado_flotante_maintenance = {
			icon_type = building_type
			icon = commerce_center
			fish = 0.08
			cloth = 0.06
			salt = 0.04
			category = building_maintenance
		}
	}

	construction_demand = town_building_construction
}
