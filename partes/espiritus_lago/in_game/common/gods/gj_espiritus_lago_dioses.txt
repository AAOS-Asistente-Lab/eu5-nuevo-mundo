# Los cuatro espiritus mayores del lago. Cada uno ofrece tres presagios.
# Un presagio es un bonus temporal que se invoca con la accion gj_espiritus_lago_select_omen.

# DIP
gj_espiritus_lago_madre_ahogada_god = {
	is_female = yes
	ability = DIP

	religion = {
		religion = gj_espiritus_lago
		name_key = gj_espiritus_lago_madre_ahogada_god
	}

	omens = {
		gj_espiritus_lago_omen_red_colmada = {
			god = gj_espiritus_lago_madre_ahogada_god

			country_modifier = {
				global_fish_output_modifier = 0.2
				global_merchant_capacity_modifier = 0.1
			}
		}
		gj_espiritus_lago_omen_aguas_hondas = {
			god = gj_espiritus_lago_madre_ahogada_god

			country_modifier = {
				naval_morale_modifier = 0.1

		}
		gj_espiritus_lago_omen_cuenta_saldada = {
			god = gj_espiritus_lago_madre_ahogada_god

			country_modifier = {
				export_efficiency = 0.1

}
