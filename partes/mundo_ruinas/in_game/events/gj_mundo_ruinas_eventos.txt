# .3 Un artefacto. Cada artefacto existe una sola vez en el mundo; si ya no queda ninguno
# del tipo de la ruina, solo hay una reliquia sin poder.
gj_mundo_ruinas.3 = {
	type = country_event
	title = gj_mundo_ruinas.3.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { has_variable = gj_art_astrolabio } }
				desc = gj_mundo_ruinas.3.desc.astrolabio
			}
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { has_variable = gj_art_cetro } }
				desc = gj_mundo_ruinas.3.desc.cetro
			}
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { has_variable = gj_art_martillo } }
				desc = gj_mundo_ruinas.3.desc.martillo
			}
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { has_variable = gj_art_cuerno } }
				desc = gj_mundo_ruinas.3.desc.cuerno
			}
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { has_variable = gj_art_escudo } }
				desc = gj_mundo_ruinas.3.desc.escudo
			}
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { has_variable = gj_art_tablillas } }
				desc = gj_mundo_ruinas.3.desc.tablillas
			}
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { has_variable = gj_art_corona } }
				desc = gj_mundo_ruinas.3.desc.corona
			}
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { has_variable = gj_art_lampara } }
				desc = gj_mundo_ruinas.3.desc.lampara
			}
			triggered_desc = {
				desc = gj_mundo_ruinas.3.desc.reliquia
			}
		}
	}
	outcome = positive

	immediate = {
		scope:gj_ruina_loc = {
			if = {
				limit = { gj_ruina_es_atlante = yes }
				random_list = {
					50 = {
						modifier = { add = -50 has_global_variable = gj_art_astrolabio_hallado }
						set_variable = gj_art_astrolabio
						root = { set_global_variable = gj_art_astrolabio_hallado }
					}
					50 = {
						modifier = { add = -50 has_global_variable = gj_art_cetro_hallado }
						set_variable = gj_art_cetro
						root = { set_global_variable = gj_art_cetro_hallado }
					}
					1 = { }
				}
			}
			else_if = {
				limit = { gj_ruina_es_enana = yes }
				random_list = {
					34 = {
						modifier = { add = -34 has_global_variable = gj_art_martillo_hallado }
						set_variable = gj_art_martillo
						root = { set_global_variable = gj_art_martillo_hallado }
					}
					33 = {
						modifier = { add = -33 has_global_variable = gj_art_cuerno_hallado }
						set_variable = gj_art_cuerno
						root = { set_global_variable = gj_art_cuerno_hallado }
					}
					33 = {
						modifier = { add = -33 has_global_variable = gj_art_escudo_hallado }
						set_variable = gj_art_escudo
						root = { set_global_variable = gj_art_escudo_hallado }
					}
					1 = { }
				}
			}
			else = {
				random_list = {
					34 = {
						modifier = { add = -34 has_global_variable = gj_art_tablillas_hallado }
						set_variable = gj_art_tablillas
						root = { set_global_variable = gj_art_tablillas_hallado }
					}
					33 = {
						modifier = { add = -33 has_global_variable = gj_art_corona_hallado }
						set_variable = gj_art_corona
						root = { set_global_variable = gj_art_corona_hallado }
					}
					33 = {
						modifier = { add = -33 has_global_variable = gj_art_lampara_hallado }
						set_variable = gj_art_lampara
						root = { set_global_variable = gj_art_lampara_hallado }
					}
					1 = { }
				}
			}
			gj_ruinas_cierra_ruina = yes
		}
		gj_ruinas_sincroniza_artefactos = yes
	}

	# Custodiarlo en la ruina: queda donde esta (y lo hereda quien conquiste esa tierra).
	option = {
		name = gj_mundo_ruinas.3.a
		ai_chance = { base = 50 }
		add_prestige = prestige_weak_bonus
		change_gold_effect = { scale = 1 }
	}
	# Llevarlo a la capital: queda donde esta la corte (y se pierde si cae la capital).
	option = {
		name = gj_mundo_ruinas.3.b
		ai_chance = { base = 50 }
		gj_ruinas_mueve_artefactos = yes
		gj_ruinas_sincroniza_artefactos = yes
		add_legitimacy = legitimacy_weak_bonus
	}
}

# .4 Saber perdido: un avance de la era siguiente, antes de tiempo.
gj_mundo_ruinas.4 = {
	type = country_event
	title = gj_mundo_ruinas.4.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { gj_ruina_es_atlante = yes } }
				desc = gj_mundo_ruinas.4.desc.atlante
			}
			triggered_desc = {
				trigger = { scope:gj_ruina_loc = { gj_ruina_es_enana = yes } }
				desc = gj_mundo_ruinas.4.desc.enana
			}
			triggered_desc = {
				desc = gj_mundo_ruinas.4.desc.antigua
			}
		}
	}
	outcome = positive

	immediate = {
		scope:gj_ruina_loc = { gj_ruinas_cierra_ruina = yes }
	}

	option = {
		name = gj_mundo_ruinas.4.a
		ai_chance = { base = 70 }
		add_country_modifier = { modifier = gj_saber_perdido_impulso years = 10 }
		if = {
			limit = { scope:gj_ruina_loc = { gj_ruina_es_atlante = yes } }
			if = {
				limit = { current_age_or_later = { age = age_3_discovery } }
				unlock_advance_effect = { type = gj_saber_cartas_abisales }
			}
			else = {
				unlock_advance_effect = { type = gj_saber_mareas_hundidas }
			}
		}
		else_if = {
			limit = { scope:gj_ruina_loc = { gj_ruina_es_enana = yes } }
			if = {
				limit = { current_age_or_later = { age = age_3_discovery } }
				unlock_advance_effect = { type = gj_saber_runas_de_carga }
			}
			else = {
				unlock_advance_effect = { type = gj_saber_forja_cantora }
			}
		}
		else = {
			if = {
				limit = { current_age_or_later = { age = age_3_discovery } }
				unlock_advance_effect = { type = gj_saber_calendario_primero }
			}
			else = {
				unlock_advance_effect = { type = gj_saber_riego_anterior }
			}
		}
	}
	option = {
		name = gj_mundo_ruinas.4.b
		ai_chance = { base = 30 }
		change_gold_effect = { scale = 12 }
	}
}

# .8 La expedicion no regresa: puede morir quien la dirigia.
gj_mundo_ruinas.8 = {
	type = country_event
	title = gj_mundo_ruinas.8.title
	desc = gj_mundo_ruinas.8.desc
	outcome = negative

	immediate = {
		random_character = {
			limit = {
				is_ruler = no
				is_courtier = yes
			}
			save_scope_as = gj_lider
		}
		if = {
			limit = { exists = scope:gj_lider }
			kill_character_silently = scope:gj_lider
		}
	}

	option = {
		name = gj_mundo_ruinas.8.a
		ai_chance = { base = 50 }
		add_prestige = prestige_weak_penalty
	}
	option = {
		name = gj_mundo_ruinas.8.b
		trigger = {
			gold > { value = monthly_income_trade_and_tax multiply = 2 }
		}
		ai_chance = { base = 50 }
		change_gold_effect = { scale = -3 }
		remove_variable = gj_expedicion_cd
	}
}
