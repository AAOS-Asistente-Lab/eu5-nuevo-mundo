namespace = gj_luz_fundadores

# ------------------------------------------------------------
# La Luz de los Fundadores: 12 eventos. root = pais seguidor.
# Se disparan desde gj_luz_fundadores_pulso (1, 4-8, 10, 12), desde la muerte del soberano (2),
# al canonizar (3), al ascender una villa (9, desde el efecto de contabilidad) y al leer un presagio (11).
# ------------------------------------------------------------

# .1 La primera lumbre de la ciudad
gj_luz_fundadores.1 = {
	type = country_event
	title = gj_luz_fundadores.1.title
	desc = gj_luz_fundadores.1.desc
	outcome = positive

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	trigger = {
		gj_luz_fundadores_sigue = yes
		at_war = no
		any_owned_location = {
			is_city = yes
		}
	}

	immediate = {
		random_owned_location = {
			limit = {
				is_city = yes
			}
			save_scope_as = gj_ciudad
		}
	}

	option = {
		name = gj_luz_fundadores.1.a
		scope:gj_ciudad = {
			add_location_modifier = {
				modifier = gj_hogar_consagrado_modifier
				years = modifier_duration_years_short
				mode = add
			}
		}
		add_religious_influence = religious_influence_weak_bonus
	}

	option = {
		name = gj_luz_fundadores.1.b
		add_stability = stability_weak_bonus
	}
}

# .2 Los altares piden un Fundador (muere un soberano con 15 puntos de fundador)
gj_luz_fundadores.2 = {
	type = country_event
	title = gj_luz_fundadores.2.title
	desc = gj_luz_fundadores.2.desc
	outcome = neutral

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	trigger = {
		gj_luz_fundadores_sigue = yes
		scope:old_ruler = {
			has_variable = gj_fundador_puntos
			var:gj_fundador_puntos >= 15
		}
	}

	immediate = {
		scope:old_ruler = { save_scope_as = gj_difunto }
	}

	option = {
		name = gj_luz_fundadores.2.a
		add_religious_influence = religious_influence_severe_bonus
		add_legitimacy = legitimacy_weak_bonus
		custom_tooltip = gj_luz_fundadores_canonizar_tt
	}

	option = {
		name = gj_luz_fundadores.2.b
		add_prestige = prestige_weak_bonus
		add_stability = stability_weak_bonus
	}
}

# .3 La ciudad del Fundador (se canoniza a un santo de la Luz)
gj_luz_fundadores.3 = {
	type = country_event
	title = gj_luz_fundadores.3.title
	desc = gj_luz_fundadores.3.desc
	outcome = positive

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	trigger = {
		gj_luz_fundadores_sigue = yes
		any_owned_location = {
			is_city = yes
		}
	}

	immediate = {
		scope:recipient = { save_scope_as = gj_santo }
		ordered_owned_location = {
			limit = {
				is_city = yes
				NOT = { has_variable = gj_ciudad_del_fundador }
			}
			order_by = population
			max = 1
			check_range_bounds = no
			save_scope_as = gj_ciudad
		}
	}

	option = {
		name = gj_luz_fundadores.3.a
		if = {
			limit = {
				exists = scope:gj_ciudad
			}
			scope:gj_ciudad = {
				add_location_modifier = {
					modifier = gj_ciudad_del_fundador_modifier
					years = modifier_duration_years_century
					mode = add
				}
				set_variable = gj_ciudad_del_fundador
			}
		}
		add_legitimacy = legitimacy_mild_bonus
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
	}

	option = {
		name = gj_luz_fundadores.3.b
		every_owned_location = {
			limit = {
				location_rank = location_rank:town
			}
			add_location_modifier = {
				modifier = gj_hogar_consagrado_modifier
				years = modifier_duration_years_short
				mode = add
			}
		}
		add_prestige = prestige_weak_bonus
	}
}
