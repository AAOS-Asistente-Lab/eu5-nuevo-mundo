gj_skarren_incursion = {
	decision_category = country_specific_decisions

	potential = {
		gj_es_skarren = yes
	}

	allow = {
		any_neighbor_country = { is_subject = no }
		NOT = { has_variable = gj_skarren_incursion_enfriamiento }
	}

	ai_will_do = { value = 60 }

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	option = {
		name = gj_skarren_incursion.a
		ai_chance = { value = 100 }
		effect = {
			random_neighbor_country = {
				limit = { is_subject = no }
				save_scope_as = gj_victima
				add_country_modifier = { modifier = gj_skarren_costa_saqueada years = 5 mode = add_and_extend }
			}
			change_gold_effect = { scale = 3 }
			add_prestige = prestige_weak_bonus
			set_variable = { name = gj_skarren_botin_reciente value = yes years = 6 }
			set_variable = { name = gj_skarren_incursion_enfriamiento value = yes years = 2 }
			gj_skarren_suma_gloria = { N = 3 }
		}
	}
}
