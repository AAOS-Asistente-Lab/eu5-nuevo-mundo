# Ganchos de la mecanica velathi. La integracion los enlaza a las on_actions del juego:
#   on_new_ruler                       -> gj_velathi_nuevo_rey
#   yearly_country_pulse               -> gj_velathi_yearly_pulse
#   on_took_location_in_peace_treaty   -> gj_velathi_conquista

gj_velathi_nuevo_rey = {
	trigger = {
		gj_es_velathi = yes
	}
	effect = {
		set_variable = { name = gj_velathi_anyos_reinado value = 0 }
		gj_velathi_reinicia_cansancio_effect = yes
		if = {
			limit = { has_country_modifier = gj_velathi_consejo_regente }
			remove_country_modifier = gj_velathi_consejo_regente
		}
	}
	events = {
		delay = { days = { 20 60 } }
		gj_velathi.1
	}
}

gj_velathi_yearly_pulse = {
	trigger = {
		gj_es_velathi = yes
	}
	effect = {
		if = {
			limit = { NOT = { has_variable = gj_velathi_anyos_reinado } }
			set_variable = { name = gj_velathi_anyos_reinado value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = gj_velathi_cansancio } }
			set_variable = { name = gj_velathi_cansancio value = 0 }
		}
		change_variable = { name = gj_velathi_anyos_reinado add = 1 }
	}
	events = {
		gj_velathi.2
	}
	random_events = {
		chance_to_happen = 14
		100 = 0
		12 = gj_velathi.3
		10 = gj_velathi.6
		10 = gj_velathi.7
		8 = gj_velathi.8
		10 = gj_velathi.9
		8 = gj_velathi.10
		8 = gj_velathi.11
		8 = gj_velathi.12
		8 = gj_velathi.13
		8 = gj_velathi.15
		8 = gj_velathi.17
		8 = gj_velathi.18
		6 = gj_velathi.19
		6 = gj_velathi.20
	}
}

gj_velathi_conquista = {
	trigger = {
		exists = scope:winner
		scope:winner = { gj_es_velathi = yes }
	}
	effect = {
		scope:winner = {
			if = {
				limit = { NOT = { has_variable = gj_velathi_conquistas } }
				set_variable = { name = gj_velathi_conquistas value = 0 }
			}
			change_variable = { name = gj_velathi_conquistas add = 1 }
			gj_velathi_karma_penalty_effect = yes
			if = {
				limit = { var:gj_velathi_conquistas >= 3 }
				set_variable = { name = gj_velathi_conquistas value = 0 }
				trigger_event_non_silently = { id = gj_velathi.5 }
			}
		}
	}
}
