namespace = gj_llama_profunda

# ------------------------------------------------------------
# Eventos de la Llama Profunda (fe de Durnhal).
# Los lanza el pulso anual (on_action gj_llama_profunda_pulso_anual), salvo el .4 y el .11, que los programan
# la decision del pacto y el efecto de romperlo, y el .20, que es oculto.
# ------------------------------------------------------------

# .1 El Encendido de la Primera Forja (fiesta, pureza media o alta)
gj_llama_profunda.1 = {
	type = country_event
	title = gj_llama_profunda.1.title
	desc = gj_llama_profunda.1.desc
	outcome = positive

	illustration_tags = {
		10 = happy
		10 = interior
	}

	trigger = {
		religion = religion:gj_llama_profunda
		purity >= 30
		at_war = no
		NOT = { has_variable = gj_llama_encendido_enfriamiento }
	}

	immediate = {
		set_variable = { name = gj_llama_encendido_enfriamiento value = 1 years = 10 }
	}

	option = {
		name = gj_llama_profunda.1.a
		add_gold = -40
		add_prestige = prestige_mild_bonus
		add_purity = purity_mild_bonus
	}

	option = {
		name = gj_llama_profunda.1.b
		add_gold = -20
		add_country_modifier = { modifier = gj_llama_forja_bendita years = 8 mode = add_and_extend }
	}

	option = {
		name = gj_llama_profunda.1.c
		add_stability = stability_weak_bonus
		add_purity = purity_weak_bonus
	}
}

# .4 El Diezmo del Ala (cada diez anos, mientras dure el pacto con el dragon)
gj_llama_profunda.4 = {
	type = country_event
	title = gj_llama_profunda.4.title
	desc = gj_llama_profunda.4.desc
	outcome = negative

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	trigger = {
		religion = religion:gj_llama_profunda
		has_variable = gj_llama_pacto_dragon
	}

	option = {
		name = gj_llama_profunda.4.a
		random_owned_location = {
			every_pop = { add_pop_size = { value = pop_size multiply = -0.08 } }
		}
		random_owned_location = {
			every_pop = { add_pop_size = { value = pop_size multiply = -0.08 } }
		}
		random_owned_location = {
			every_pop = { add_pop_size = { value = pop_size multiply = -0.08 } }
		}
		add_purity = purity_weak_penalty
		add_country_modifier = { modifier = gj_llama_dragon_pacto_cumplido years = 10 mode = add_and_extend }
		trigger_event_silently = { id = gj_llama_profunda.4 days = 3650 }
	}

	option = {
		name = gj_llama_profunda.4.b
		trigger = { gold >= 250 }
		add_gold = -250
		add_country_modifier = { modifier = gj_llama_dragon_pacto_cumplido years = 10 mode = add_and_extend }
		trigger_event_silently = { id = gj_llama_profunda.4 days = 3650 }
	}

	option = {
		name = gj_llama_profunda.4.c
		gj_llama_profunda_rompe_pacto = yes
	}
}
