# Efectos de la Llama Profunda. El ambito es siempre un pais.

# Como lee esta fe un cometa o un eclipse (lo llama mundo_bestias): una chispa del fuego de abajo.
# Mas forja y mas estabilidad; si el pais sigue la Llama Profunda, ademas sube la pureza.
gj_llama_profunda_lee_presagio = {
	add_stability = stability_weak_bonus
	add_prestige = prestige_weak_bonus
	add_country_modifier = { modifier = gj_llama_chispa_de_abajo years = 5 mode = add_and_extend }
	if = {
		limit = {
			religion = religion:gj_llama_profunda
		}
		add_purity = purity_weak_bonus
	}
}

# Baja la inquietud de la profundidad. La variable es de la parte durnhal: aqui no se escribe.
# Se llama a su efecto con puntos negativos; el suelo en 0 lo pone durnhal.
gj_llama_profunda_baja_inquietud = {
	gj_durnhal_sube_inquietud = { PUNTOS = -$PUNTOS$ }
}

# Sube la inquietud de la profundidad llamando al efecto de durnhal (su dueno).
gj_llama_profunda_sube_inquietud = {
	gj_durnhal_sube_inquietud = { PUNTOS = $PUNTOS$ }
}

# Da al pais el panteon de partida (cinco dioses). Se usa por si el grupo de la fe no cuenta como pagano
# y el juego no los reparte solo. No quita los que ya tenga.
gj_llama_profunda_da_panteon = {
	add_god = god:gj_llama_orgad_god
	add_god = god:gj_llama_thurn_god
	add_god = god:gj_llama_veshka_god
	add_god = god:gj_llama_ildra_god
	add_god = god:gj_llama_maruk_god
	set_variable = { name = gj_llama_panteon_dado value = 1 }
}

# El pais rompe el pacto con la Brasa Mayor: pierde al dios, la pureza y el favor del dragon.
gj_llama_profunda_rompe_pacto = {
	remove_god = god:gj_llama_zhurgan_god
	remove_variable = gj_llama_pacto_dragon
	add_purity = purity_mild_bonus
	add_country_modifier = { modifier = gj_llama_dragon_rencor years = 10 mode = add_and_extend }
	trigger_event_silently = { id = gj_llama_profunda.11 days = 120 }
	if = {
		limit = {
			has_global_variable = gj_dragon_estado
			global_var:gj_dragon_estado < 2
		}
		set_global_variable = {
			name = gj_dragon_estado
			value = {
				value = global_var:gj_dragon_estado
				add = 1
			}
		}
	}
}
