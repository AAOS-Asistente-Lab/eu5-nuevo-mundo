# Decisiones de Durnhal

gj_durnhal_cavar_mas_hondo = {
	decision_category = economy_decisions

	potential = {
		gj_es_durnhal = yes
	}

	allow = {
		NOT = { has_country_modifier = gj_durnhal_obra_en_curso }
		political_influence >= gj_durnhal_coste_cavar
		any_owned_location = { gj_durnhal_puede_profundizar = yes }
	}

	ai_will_do = {
		value = 50
	}

	option = {
		name = gj_durnhal_cavar_mas_hondo.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = {
				value = gj_durnhal_coste_cavar
				multiply = -1
			}
			ordered_owned_location = {
				limit = { gj_durnhal_puede_profundizar = yes }
				order_by = population
				max = 1
				gj_durnhal_profundizar_efecto = yes
			}
			gj_durnhal_cambia_inquietud = { valor = 6 }
			add_country_modifier = { modifier = gj_durnhal_obra_en_curso years = 2 mode = add_and_extend }
		}
	}
}
