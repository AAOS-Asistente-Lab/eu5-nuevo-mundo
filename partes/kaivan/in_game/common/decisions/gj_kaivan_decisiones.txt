# La Gran Cabalgada: los clanes se reunen diez anos. Mas velocidad y saqueo; despues la hierba cuesta volver.
gj_kaivan_gran_cabalgada_decision = {
	decision_category = country_specific_decisions

	potential = {
		gj_es_kaivan = yes
	}

	allow = {
		is_subject = no
		NOT = { has_variable = gj_kaivan_cabalgada_enfriamiento }
		NOT = { has_variable = gj_kaivan_disputa_abierta }
		NOT = { has_country_modifier = gj_kaivan_gran_cabalgada }
	}

	ai_will_do = {
		value = 20
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	option = {
		name = gj_kaivan_gran_cabalgada_decision.a
		ai_chance = { value = 100 }
		effect = {
			add_country_modifier = {
				modifier = gj_kaivan_gran_cabalgada
				years = 10
				mode = add
			}
			add_prestige = prestige_weak_bonus
			set_variable = { name = gj_kaivan_cabalgada_enfriamiento value = 1 days = 18250 }
			trigger_event_non_silently = { id = gj_kaivan.48 days = 3650 }
		}
	}
}
