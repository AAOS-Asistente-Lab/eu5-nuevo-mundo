#######################################################################
# 1. El Circulo se reune
#######################################################################
gj_kaivan.1 = {
	type = country_event
	title = gj_kaivan.1.title
	desc = gj_kaivan.1.desc
	outcome = neutral

	illustration_tags = {
		10 = regular
		10 = interior
	}

	trigger = {
		gj_es_kaivan = yes
		is_subject = no
		NOT = { has_variable = gj_kaivan_disputa_abierta }
	}

	immediate = {
		gj_kaivan_prepara_circulo = yes
	}

	option = {
		name = gj_kaivan.1.a
		trigger = {
			any_owned_location = { gj_zona_estepa_alta = yes }
		}
		ai_chance = { base = 25 }
		gj_kaivan_apoya_clan = { C = uzhar }
		add_gold = { value = root.monthly_income_trade_and_tax multiply = -1 }
		gj_kaivan_resuelve_circulo = yes
		trigger_event_non_silently = gj_kaivan.2
	}
	option = {
		name = gj_kaivan.1.b
		trigger = {
			any_owned_location = { gj_zona_marca_occidental = yes }
		}
		ai_chance = { base = 25 }
		gj_kaivan_apoya_clan = { C = tarkhen }
		add_gold = { value = root.monthly_income_trade_and_tax multiply = -1 }
		gj_kaivan_resuelve_circulo = yes
		trigger_event_non_silently = gj_kaivan.2
	}
	option = {
		name = gj_kaivan.1.c
		trigger = {
			any_owned_location = { gj_zona_valles_orientales = yes }
		}
		ai_chance = { base = 25 }
		gj_kaivan_apoya_clan = { C = ilbari }
		add_gold = { value = root.monthly_income_trade_and_tax multiply = -1 }
		gj_kaivan_resuelve_circulo = yes
		trigger_event_non_silently = gj_kaivan.2
	}
	option = {
		name = gj_kaivan.1.d
		trigger = {
			any_owned_location = { gj_zona_estepa_baja = yes }
		}
		ai_chance = { base = 25 }
		gj_kaivan_apoya_clan = { C = merghan }
		add_gold = { value = root.monthly_income_trade_and_tax multiply = -1 }
		gj_kaivan_resuelve_circulo = yes
		trigger_event_non_silently = gj_kaivan.2
	}
	option = {
		name = gj_kaivan.1.e
		fallback = yes
		ai_chance = { base = 5 }
		gj_kaivan_resuelve_circulo = yes
		trigger_event_non_silently = gj_kaivan.2
	}
}

#######################################################################
# 2. La tienda del nuevo kan (resultado de la votacion)
#######################################################################
gj_kaivan.2 = {
	type = country_event
	title = gj_kaivan.2.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:gj_kaivan_ganador = 1 }
				desc = gj_kaivan.2.desc.uzhar
			}
			triggered_desc = {
				trigger = { var:gj_kaivan_ganador = 2 }
				desc = gj_kaivan.2.desc.tarkhen
			}
			triggered_desc = {
				trigger = { var:gj_kaivan_ganador = 3 }
				desc = gj_kaivan.2.desc.ilbari
			}
			triggered_desc = {
				trigger = { var:gj_kaivan_ganador = 4 }
				desc = gj_kaivan.2.desc.merghan
			}
			triggered_desc = {
				desc = gj_kaivan.2.desc.nadie
			}
		}
	}
	outcome = positive

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		gj_kaivan_aplica_reinado_del_ganador = yes
	}

	option = {
		name = gj_kaivan.2.a
		add_prestige = prestige_mild_bonus
		add_legitimacy = legitimacy_weak_bonus
		if = {
			limit = { var:gj_kaivan_rebelde > 0 }
			trigger_event_non_silently = { id = gj_kaivan.3 days = 20 }
		}
	}
}

# 3. Los clanes agraviados (votacion renida)

gj_kaivan.3 = {
	type = country_event
	title = gj_kaivan.3.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:gj_kaivan_rebelde = 1 }
				desc = gj_kaivan.3.desc.uzhar
			}
			triggered_desc = {
				trigger = { var:gj_kaivan_rebelde = 2 }
				desc = gj_kaivan.3.desc.tarkhen
			}
			triggered_desc = {
				trigger = { var:gj_kaivan_rebelde = 3 }
				desc = gj_kaivan.3.desc.ilbari
			}
			triggered_desc = {
				desc = gj_kaivan.3.desc.merghan
			}
		}
	}
	outcome = negative

	illustration_tags = {
		10 = regular
		10 = interior
	}

	trigger = {
		gj_es_kaivan = yes
		var:gj_kaivan_rebelde > 0
		NOT = { has_variable = gj_kaivan_disputa_abierta }
	}

	option = {
		name = gj_kaivan.3.a
		ai_chance = { base = 50 }
		add_gold = { value = root.monthly_income_trade_and_tax multiply = -4 }
		if = {
			limit = { country_has_estate = estate_type:tribes_estate }
			add_estate_satisfaction = { type = estate_type:tribes_estate value = estate_satisfaction_mild_bonus }
		}
		add_legitimacy = legitimacy_weak_penalty
		set_variable = { name = gj_kaivan_rebelde value = 0 }
	}
	option = {
		name = gj_kaivan.3.b
		ai_chance = { base = 15 }
		gj_kaivan_secesion_de_clan = yes
		add_prestige = prestige_mild_penalty
		set_variable = { name = gj_kaivan_rebelde value = 0 }
	}
	option = {
		name = gj_kaivan.3.c
		ai_chance = { base = 35 }
		set_variable = { name = gj_kaivan_disputa_abierta value = 1 }
	}
}
