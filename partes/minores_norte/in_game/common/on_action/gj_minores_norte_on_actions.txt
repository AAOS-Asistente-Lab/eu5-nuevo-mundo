# Pulso anual de los menores del norte. Se cuelga de yearly_country_pulse (ver el fichero de enganche).
# Cada sub-on-action tiene un disparador barato: los paises que no son de su estilo no pagan nada.

gj_minores_norte_pulso_anual = {
	trigger = {
		gj_minores_norte_menor_norte = yes
	}
	on_actions = {
		gj_minores_norte_pulso_ciudad_estado
		gj_minores_norte_pulso_clan
		gj_minores_norte_pulso_marca
		gj_minores_norte_pulso_propios
	}
}

gj_minores_norte_pulso_ciudad_estado = {
	trigger = {
		gj_minores_norte_menor_ciudad_estado = yes
	}
	random_events = {
		chance_to_happen = 18
		100 = gj_minores_norte.1
		100 = gj_minores_norte.2
		100 = gj_minores_norte.3
		100 = gj_minores_norte.4
		100 = gj_minores_norte.5
		100 = gj_minores_norte.6
	}
}
