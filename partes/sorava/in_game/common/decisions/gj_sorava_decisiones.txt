# Sorava: decisiones de «La palabra de las Casas» y del estrecho.
# Los tres contratos se firman por cinco anos; al cerrarse llega un evento donde se cumple o se rompe la palabra dada.

gj_sorava_contrato_especias = {
	decision_category = economy_decisions

	potential = {
		gj_es_sorava = yes
	}

	allow = {
		NOT = { has_country_modifier = gj_sorava_contrato_especias }
		at_war = no
		gold > {
			value = monthly_income_trade_and_tax
			multiply = 3
		}
	}

	ai_will_do = {
		value = 40
	}

	option = {
		name = gj_sorava_contrato_especias.a
		ai_chance = { value = 100 }
		effect = {
			change_gold_effect = { scale = -3 }
			add_country_modifier = {
				modifier = gj_sorava_contrato_especias
				years = 5
				mode = add
			}
			trigger_event_non_silently = {
				id = gj_sorava.21
				months = 60
			}
		}
	}
}

# Disputar el estrecho a la Liga: una apuesta naval de cinco anos. Pide tener costa en la zona del estrecho.
gj_sorava_disputar_estrecho = {
	decision_category = expansion_decisions

	potential = {
		gj_es_sorava = yes
	}

	allow = {
		NOT = { has_country_modifier = gj_sorava_pulso_estrecho }
		any_owned_location = { gj_zona_estrecho = yes }
		gold > {
			value = monthly_income_trade_and_tax
			multiply = 5
		}
	}

	ai_will_do = {
		value = 25
	}

	option = {
		name = gj_sorava_disputar_estrecho.a
		ai_chance = { value = 100 }
		effect = {
			change_gold_effect = { scale = -5 }
			add_country_modifier = {
				modifier = gj_sorava_pulso_estrecho
				years = 5
				mode = add
			}
			add_prestige = prestige_mild_bonus
			gj_sorava_cambiar_palabra = { valor = -5 }
			every_country = {
				limit = { gj_es_lagunas = yes }
				trigger_event_non_silently = gj_sorava.30
			}
		}
	}
}

# La Lonja Mayor: se funda una sola vez, cuando la capital ya tiene su Sala de las Balanzas.
gj_sorava_lonja_mayor = {
	decision_category = economy_decisions

	potential = {
		gj_es_sorava = yes
		current_age_or_later = { age = age_2_renaissance }
	}

	allow = {
		capital = {
			has_building = building_type:gj_sorava_sala_de_balanzas
		}
		gold > {
			value = monthly_income_trade_and_tax
			multiply = 8
		}
	}

	only_once = yes

	ai_will_do = {
		value = 100
	}

	option = {
		name = gj_sorava_lonja_mayor.a
		ai_chance = { value = 100 }
		effect = {
			change_gold_effect = { scale = -8 }
			add_country_modifier = {
				modifier = gj_sorava_lonja_mayor
				years = -1
				mode = add
			}
			gj_sorava_cambiar_palabra = { valor = 10 }
			add_prestige = prestige_severe_bonus
		}
	}
}

# Auditar las Casas: limpiar los libros cuando la palabra anda baja. Se puede repetir cada diez anos.
gj_sorava_auditar_las_casas = {
	decision_category = government_decisions

	potential = {
		gj_es_sorava = yes
		has_variable = gj_sorava_palabra
	}

	allow = {
		var:gj_sorava_palabra < 60
		NOT = { has_country_modifier = gj_sorava_cuentas_claras }
		gold > {
			value = monthly_income_trade_and_tax
			multiply = 2
		}
	}

	ai_will_do = {
		value = 60
	}

	option = {
		name = gj_sorava_auditar_las_casas.a
		ai_chance = { value = 100 }
		effect = {
			change_gold_effect = { scale = -2 }
			gj_sorava_cambiar_palabra = { valor = 15 }
			add_country_modifier = {
				modifier = gj_sorava_cuentas_claras
				years = 10
				mode = add
			}
		}
	}
}

# Puerto Franco: la capital se declara puerto libre de aduanas. Una vez. Atrae contrabando: cuesta palabra.
gj_sorava_puerto_franco = {
	decision_category = economy_decisions

	potential = {
		gj_es_sorava = yes
		current_age_or_later = { age = age_3_discovery }
	}

	allow = {
		capital = {
			is_port = yes
		}
		stability >= 20
	}

	only_once = yes

	ai_will_do = {
		value = 80
	}

	option = {
		name = gj_sorava_puerto_franco.a
		ai_chance = { value = 100 }
		effect = {
			capital = {
				add_location_modifier = {
					modifier = gj_sorava_puerto_franco
					years = -1
					mode = add
				}
			}
			gj_sorava_cambiar_palabra = { valor = -8 }
			add_stability = stability_weak_penalty
		}
	}
}
