namespace = gj_sorava

# =========================================================
# Sorava: eventos de comercio. Los .1 a .19 son de sabor y salen del pulso anual (gj_sorava_pulso_anual).
# Los .21 a .23 cierran los contratos, el .30 llega a la Liga, y del .40 al .43 son de la Quiebra de las Casas.
# La palabra (gj_sorava_palabra) es la reputacion de las Casas: ver scripted_effects.
# =========================================================

# .1 El juramento de la Lonja (arranque, una sola vez)
gj_sorava.1 = {
	type = country_event
	title = gj_sorava.1.title
	desc = gj_sorava.1.desc
	outcome = positive
	fire_only_once = yes

	illustration_tags = {
		10 = regular
		10 = interior
	}

	trigger = {
		gj_es_sorava = yes
	}

	immediate = {
		set_variable = {
			name = gj_sorava_palabra
			value = 50
		}
	}

	option = {
		name = gj_sorava.1.a
		add_prestige = prestige_mild_bonus
		gj_sorava_cambiar_palabra = { valor = 5 }
	}
	option = {
		name = gj_sorava.1.b
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
	}
}
