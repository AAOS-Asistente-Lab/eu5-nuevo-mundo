# Ganchos de los menores del sur. La integracion enlaza gj_minores_sur_pulso_anual a la lista de on_actions de yearly_country_pulse.
# Un unico pulso, con tres listas segun el estilo: asi cada pais solo sortea los eventos que le tocan.
# Cada año hay una probabilidad de evento; los eventos de «una sola vez» se agotan y los repetibles siguen saliendo.

gj_minores_sur_pulso_anual = {
	trigger = {
		gj_minores_sur_es_menor = yes
	}
	on_actions = {
		gj_minores_sur_pulso_ciudad
		gj_minores_sur_pulso_selva
		gj_minores_sur_pulso_marca
	}
}

gj_minores_sur_pulso_ciudad = {
	trigger = {
		gj_minores_sur_estilo_ciudad = yes
	}
	random_events = {
		chance_to_happen = 14
		100 = 0
		12 = gj_minores_sur.1
		10 = gj_minores_sur.2
		8 = gj_minores_sur.3
		8 = gj_minores_sur.4
		6 = gj_minores_sur.5
		6 = gj_minores_sur.6
		8 = gj_minores_sur.7
		8 = gj_minores_sur.8
		8 = gj_minores_sur.9
		6 = gj_minores_sur.10
		6 = gj_minores_sur.31
		6 = gj_minores_sur.32
	}
}
