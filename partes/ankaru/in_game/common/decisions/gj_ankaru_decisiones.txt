# Decisiones de Ankarú: la Cuerda del Trabajo, los Colonos del Sol, la consagración de un gnomon y el Concilio de los Cómputos.

gj_ank_cuerda_del_trabajo = {
	decision_category = country_specific_decisions

	potential = {
		gj_es_ankaru = yes
	}

	allow = {
		at_war = no
		gold >= 150
		NOT = { has_country_modifier = gj_ank_cuerda_del_trabajo }
		NOT = { has_country_modifier = gj_ank_cuerda_descanso }
	}

	ai_will_do = {
		value = 15
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	option = {
		name = gj_ank_cuerda_del_trabajo.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = { value = -150 }
			add_country_modifier = { modifier = gj_ank_cuerda_del_trabajo years = 8 mode = replace }
			hidden_effect = {
				add_country_modifier = { modifier = gj_ank_cuerda_descanso years = 20 mode = replace }
				trigger_event_non_silently = { id = gj_ankaru.21 years = 8 }
			}
		}
	}
}

gj_ank_colonos_del_sol = {
	decision_category = country_specific_decisions

	potential = {
		gj_es_ankaru = yes
		current_age_or_later = { age = age_2_renaissance }
	}

	allow = {
		gold >= 100
		NOT = { has_country_modifier = gj_ank_colonos_del_sol }
		any_owned_location = {
			OR = {
				location_rank = location_rank:city
				location_rank = location_rank:town
			}
		}
		any_owned_location = {
			location_rank = location_rank:rural_settlement
		}
	}

	ai_will_do = {
		value = 10
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	illustration_scope = {
		random_owned_location = {
			limit = {
				OR = {
					location_rank = location_rank:city
					location_rank = location_rank:town
				}
			}
			province_definition = {
				save_scope_as = origen_colonos
			}
		}
		random_owned_location = {
			limit = {
				location_rank = location_rank:rural_settlement
			}
			province_definition = {
				save_scope_as = destino_colonos
			}
		}
	}

	option = {
		name = gj_ank_colonos_del_sol.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = { value = -100 }
			add_country_modifier = { modifier = gj_ank_colonos_del_sol years = 6 mode = replace }
			if = {
				limit = {
					exists = scope:origen_colonos
					exists = scope:destino_colonos
				}
				add_migration = {
					owner ?= root
					from = scope:origen_colonos
					to = scope:destino_colonos
					culture = culture:gj_ankaruno
					amount = 0.1
					months = 60
				}
			}
		}
	}
}

				location_rank = location_rank:megalopolis
			}
		}
	}

	ai_will_do = {
		value = 50
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	option = {
		name = gj_ank_consagrar_gnomon.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = { value = -120 }
			add_prestige = prestige_weak_bonus

					cost_multiplier = 0.9

				}
			}
		}
	}
}
