gj_ankaru.9 = {
	type = country_event
	title = gj_ankaru.9.title
	desc = gj_ankaru.9.desc
	outcome = negative

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	trigger = {
		gj_es_ankaru = yes
		any_owned_location = {
			OR = {
				has_building = building_type:gj_ank_terraza_1
				has_building = building_type:gj_ank_terraza_2
				has_building = building_type:gj_ank_terraza_3
			}
		}
	}

	immediate = {
		random_owned_location = {
			limit = {
				OR = {
					has_building = building_type:gj_ank_terraza_1
					has_building = building_type:gj_ank_terraza_2
					has_building = building_type:gj_ank_terraza_3
				}
			}
			save_scope_as = target_location
		}
	}

	option = {
		name = gj_ankaru.9.a
		add_gold = { value = -60 }
	}
	option = {
		name = gj_ankaru.9.b
		if = {
			limit = { exists = scope:target_location }
			scope:target_location = {
				change_prosperity = prosperity_mild_penalty
			}
		}
	}
}

gj_ankaru.16 = {
	type = country_event
	title = gj_ankaru.16.title
	desc = gj_ankaru.16.desc
	outcome = positive

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	trigger = {
		gj_es_ankaru = yes
		any_owned_location = {
			OR = {
				location_rank = location_rank:city
				location_rank = location_rank:town
			}
		}
		any_owned_location = {
			location_rank = location_rank:rural_settlement
		}
	}

	immediate = {
		random_owned_location = {
			limit = {
				OR = {
					location_rank = location_rank:city
					location_rank = location_rank:town
				}
			}
			province_definition = {
				save_scope_as = origen_cuerda
			}
		}
		random_owned_location = {
			limit = {
				location_rank = location_rank:rural_settlement
			}
			province_definition = {
				save_scope_as = destino_cuerda
			}
		}
	}

	option = {
		name = gj_ankaru.16.a
		add_gold = { value = -40 }
		if = {
			limit = {
				exists = scope:origen_cuerda
				exists = scope:destino_cuerda
			}
			add_migration = {
				owner ?= root
				from = scope:origen_cuerda
				to = scope:destino_cuerda
				culture = culture:gj_ankaruno
				amount = 0.05
				months = 48
			}
		}
	}
	option = {
		name = gj_ankaru.16.b
		add_stability = stability_weak_bonus
	}
}
