# Los ritos de marea: siete decisiones que se pagan con Poder de Rito (rite_power).
# Las variables gj_md_* son los enfriamientos de cada rito.

gj_mar_dormido_nana = { # Cantar la nana al mar: la via barata de recuperar Poder de Rito
	decision_category = country_specific_decisions

	potential = {
		religion = religion:gj_mar_dormido
	}

	allow = {
		gold >= 80
		NOT = { has_variable = gj_md_nana }
	}

	ai_will_do = {
		value = 30
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	option = {
		name = gj_mar_dormido_nana.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = {
				value = -80
			}
			add_rite_power = 15
			set_variable = {
				name = gj_md_nana
				value = yes
				years = 5
			}
			add_country_modifier = {
				modifier = gj_mar_nana_cantada
				years = 6
				mode = add_and_extend
			}
		}
	}
}

gj_mar_dormido_levantar_ciudad = { # Rito: levantar la ciudad (una costa del pais prospera de golpe)
	decision_category = country_specific_decisions

	potential = {
		religion = religion:gj_mar_dormido
	}

	allow = {
		rite_power >= 30
		gold >= 150
		NOT = { has_variable = gj_md_ciudad }
		any_owned_location = {
			is_coastal = yes
		}
	}

	ai_will_do = {
		value = 10
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	option = {
		name = gj_mar_dormido_levantar_ciudad.a
		ai_chance = { value = 100 }
		effect = {
			add_rite_power = -30
			add_gold = {
				value = -150
			}
			set_variable = {
				name = gj_md_ciudad
				value = yes
				years = 25
			}
			random_owned_location = {
				limit = {
					is_coastal = yes
				}
				add_location_modifier = {
					mode = add
					modifier = gj_mar_ciudad_alzada
					years = 30
				}
				change_prosperity = prosperity_mild_bonus
			}
			gj_mar_registra_rito = yes
		}
	}
}

gj_mar_dormido_sangrar_marea = { # Solo marea negra: pagar al mar con oro para despertar su pulso
	decision_category = country_specific_decisions

	potential = {
		gj_mar_es_marea_negra = yes
	}

	allow = {
		gold >= 100
		NOT = { has_variable = gj_md_sangre }
	}

	ai_will_do = {
		value = 25
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	option = {
		name = gj_mar_dormido_sangrar_marea.a
		ai_chance = { value = 100 }
		effect = {
			add_gold = {
				value = -100
			}
			add_rite_power = 20
			set_variable = {
				name = gj_md_sangre
				value = yes
				years = 6
			}
			add_country_modifier = {
				modifier = gj_mar_sangre_en_la_marea
				years = 8
				mode = add_and_extend
			}
			if = {
				limit = {
					NOT = { has_variable = gj_mar_agitacion }
				}
				set_variable = { name = gj_mar_agitacion value = 0 }
			}
			change_variable = {
				name = gj_mar_agitacion
				add = 1
			}
		}
	}
}
