namespace = gj_pacto_balanza

# Eventos del Pacto de la Balanza. Los lanza la pulsacion anual de gj_pacto_balanza_pulso, salvo:
#   .9 (lo lanza la propia pulsacion cuando dura la Tregua Sagrada) y .10 (lo lanza gj_pacto_balanza_lee_presagio).

gj_pacto_balanza.1 = { # Dos casas y una balanza
	type = country_event
	title = gj_pacto_balanza.1.title
	desc = gj_pacto_balanza.1.desc
	outcome = neutral
	trigger = {
		religion = religion:gj_pacto_balanza
		country_has_estate = estate_type:burghers_estate
		country_has_estate = estate_type:nobles_estate
		honor >= 30
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
	}

	option = {
		name = gj_pacto_balanza.1.a
		add_honor = honor_weak_bonus
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
	}
	option = {
		name = gj_pacto_balanza.1.b
		add_honor = honor_weak_bonus
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_penalty }
	}
	option = {
		name = gj_pacto_balanza.1.c
		change_gold_effect = { scale = -2 }
		add_honor = honor_mild_bonus
	}
}
